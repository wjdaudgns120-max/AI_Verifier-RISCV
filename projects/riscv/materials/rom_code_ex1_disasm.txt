# rom_code_ex1.mem 디스어셈블 결과 (SW 검증용 테스트 프로그램)
# instruction_rom 이 $readmemh 로 읽는 기계어를 RV32I 명령어로 역변환한 것.
# 아래 C 코드는 기계어로부터 복원한 동등 코드임(원본 C 파일 아님).
#
#   int add(int a, int b) { return a + b; }
#   int main(void) {
#       int a = 0;        // s0-20 = 0x1000_01EC  -> BRAM word index 123
#       int sum = 0;      // s0-24 = 0x1000_01E8  -> BRAM word index 122
#       while (a <= 9) { a++; sum = add(sum, a); }
#       while (1);        // PC 0x4C 에서 정지 (jal zero,0)
#   }
#   기대 결과: dmem[122] = 55 (0+1+...+10), dmem[123] = 10
#
# 스택: sp = 0x1000_0000 + 512 = 0x1000_0200 (RAM 64KB 영역 시작 + 512B)
# PADDR = {4'b0000, addr[27:0]} 이므로 BRAM 인덱스 = (addr & 0x0FFF_FFFF) >> 2
#
